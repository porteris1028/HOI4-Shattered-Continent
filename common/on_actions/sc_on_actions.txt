on_actions = {
	on_startup = {
		effect = {
			# Only build the shattered world for the 1946 bookmark, and only once.
			if = {
				limit = {
					date > 1946.1.1
					NOT = { has_global_flag = sc_world_setup_done }
				}
				set_global_flag = sc_world_setup_done
				sc_setup_shattered_europe = yes
			}
			else_if = {
				limit = { NOT = { date > 1946.1.1 } NOT = { has_global_flag = sc_wrong_bookmark_logged } }
				set_global_flag = sc_wrong_bookmark_logged
				log = "[Shattered Continent] Not the 1946 bookmark - world setup skipped"
			}
		}
	}

	# Fires for every country once a month.
	on_monthly = {
		effect = {
			if = { limit = { tag = WST } sc_wst_monthly_tick = yes }
			else = { sc_border_raid_check = yes }

			if = { limit = { has_country_flag = sc_nomadic } sc_warband_monthly_tick = yes }
			if = { limit = { has_country_flag = sc_cult_state } sc_blo_monthly_tick = yes }
			if = { limit = { tag = BNT } sc_bnt_monthly_tick = yes }
		}
	}
}
