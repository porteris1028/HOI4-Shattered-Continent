sc_fno_category = {
	# Cheap wargoal against any European neighbour or coastal state.
	sc_fno_plan_invasion = {
		icon = generic_prepare_civil_war
		target_root_trigger = { tag = FNO }
		target_trigger = {
			FROM = {
				exists = yes
				NOT = { tag = WST }
				NOT = { tag = ROOT }
				capital_scope = { is_on_continent = europe }
				any_owned_state = {
					OR = {
						any_neighbor_state = { is_owned_by = ROOT }
						is_coastal = yes
					}
				}
			}
		}
		available = { has_war = no }
		cost = 50
		days_re_enable = 180
		complete_effect = { create_wargoal = { type = annex_everything target = FROM } }
		ai_will_do = {
			base = 2
			modifier = { factor = 5 FROM = { tag = SWE } }
		}
	}
	# Colonise wasteland that is still too dangerous for anyone else.
	sc_fno_colonize = {
		icon = generic_construction
		state_target = yes
		on_map_mode = map_and_decisions_view
		target_root_trigger = { tag = FNO }
		target_trigger = {
			FROM = {
				sc_adjacent_wasteland = yes
				check_variable = { sc_danger < 50 }
				NOT = { has_state_flag = sc_irradiated }
			}
		}
		cost = 25
		days_remove = 30
		remove_effect = { sc_wastes_claim = yes }
		ai_will_do = { base = 8 }
	}
}
