# Builds the 1946 "Shattered Continent" world on game start.

sc_setup_shattered_europe = {
	# 1. Every European state becomes empty wasteland. Vanilla countries left
	#    with no states (Germany, France, Italy, etc.) cease to exist.
	#    Countries with land outside Europe (the USSR's Asian states, colonies) survive as rumps.
	every_state = {
		limit = { is_on_continent = europe }
		WST = { transfer_state = PREV }
	}

	# 2. Hand states to the new nations. This effect is GENERATED from
	#    setup/state_assignments.csv by tools/build_setup.py - edit the CSV, not the effect.
	sc_assign_starting_states = yes

	# 3. Starting armies, spawned at each nation's capital.
	sc_spawn_starting_armies = yes

	# 4. Set up camps for the roaming warlords.
	every_country = {
		limit = { has_country_flag = sc_nomadic }
		capital_scope = {
			sc_init_forage = yes
			ROOT = { set_variable = { sc_camp_forage = PREV.sc_forage } }
		}
	}
}
