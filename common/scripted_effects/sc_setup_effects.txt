# Builds the 1946 "Shattered Continent" world on game start (called from on_startup).

sc_setup_shattered_europe = {
	# 1. Every European state not held by a spared nation becomes No Man's Land.
	#    Germany, France, Italy, the USSR in Europe, etc. cease to exist.
	every_state = {
		limit = {
			is_on_continent = europe
			NOT = { sc_is_spared_state = yes }
		}
		WST = { transfer_state = PREV }
	}

	# 2. Hand out states. GENERATED from setup/state_assignments.csv (tools/build_setup.py).
	#    This includes Japan's gains on the Pacific coast and the Siberian generals.
	sc_assign_starting_states = yes

	# 3. Whatever the USSR still holds east of the Urals and no general has claimed
	#    becomes empty taiga.
	every_state = {
		limit = { is_owned_by = SOV }
		WST = { transfer_state = PREV }
	}

	# 4. Danger: random everywhere, then the blast zones from setup/danger_zones.csv.
	sc_init_danger = yes
	sc_apply_danger_zones = yes

	# 5. Armies (GENERATED from setup/nations.toml).
	sc_spawn_starting_armies = yes

	# 6. Roaming warlords pitch their first camp.
	every_country = {
		limit = { has_country_flag = sc_nomadic }
		capital_scope = {
			sc_init_forage = yes
			ROOT = { set_variable = { sc_camp_forage = PREV.sc_forage } }
		}
	}

	# 7. Nation-specific starting situations.
	if = { limit = { country_exists = BLO } BLO = { sc_blo_setup = yes } }
	if = { limit = { country_exists = ENG } ENG = { sc_eng_setup_uprisings = yes } }
}
